Easy
C++ Runtime: 32 ms | Memory: 10.4 MB 
class Solution {
public:
    vector twoSum(vector& nums, int target) {
        unordered_map mp;
        for (int i = 0; i < nums.size(); ++i) {
            int diff = target - nums[i];
            if (mp.count(diff)) return {mp[diff], i};
            mp[nums[i]] = i;
        }
        return {};
    }
};

Direct Browser-to-GitHub Connection

Local Chrome Storage

Direct REST API Commits

Commits are made straight from your local client context to GitHub over standard HTTPS.
